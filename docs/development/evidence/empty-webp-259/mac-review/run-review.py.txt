import datetime, hashlib, json, os, pathlib, re, subprocess, sys, time, urllib.request

root=pathlib.Path(__file__).resolve().parent/'vink'
mode=sys.argv[1] if len(sys.argv)>1 else 'browser-main'
assert re.fullmatch('[a-z0-9-]+',mode)
database='vink267_review_'+mode.replace('-','_')
out=root/'target/mac-review-267'/mode
out.mkdir(parents=True,exist_ok=True)
url='postgres://vink267@127.0.0.1:55567/'+database
env=dict(os.environ,DATABASE_URL=url,VINK_MIGRATE_DATABASE_URL=url,VINK_PROFILE='local',
         VINK_DEV_PRINCIPAL='11111111-1111-4111-8111-111111111111',VINK_AUTH_SIGN_IN='disabled',
         VINK_BIND='127.0.0.1:52567',VINK_PUBLIC_ORIGIN='http://127.0.0.1:52567',
         VINK_SITE_ROOT=str(root/'target/site'),VINK_LOG_FORMAT='text')
binary=str(root/'target/debug/vink-app')
command=['node',sys.argv[3] if len(sys.argv)>3 else 'apps/vink-web/e2e/browser-smoke.mjs','--base','http://127.0.0.1:52567',
         '--chrome','/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',
         '--state',str(out/'smoke-state.json'),'--phase',sys.argv[2] if len(sys.argv)>2 else 'main']
record={'command':command,'cwd':str(root),'source_revision':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),
        'started_utc':datetime.datetime.now(datetime.UTC).isoformat(),'supervisor_pid':os.getpid(),
        'runtime_sha256':{name:hashlib.sha256((root/name).read_bytes()).hexdigest() for name in ['target/debug/vink-app','target/site/pkg/vink.js','target/site/pkg/vink_bg.wasm','target/site/pkg/vink.css']},
        'boundary':'Independent Mac browser main on exact hash-verified author artifacts copied into review checkout; private PostgreSQL cluster. Not a rebuild or xtask Web PASS.'}
(out/'process.json').write_text(json.dumps(record,indent=2)+'\n')
try:
    with (out/'setup.log').open('w') as log:
        subprocess.run(['/opt/homebrew/opt/postgresql@17/bin/createdb','-h','127.0.0.1','-p','55567','-U','vink267',database],stdout=log,stderr=subprocess.STDOUT,check=True)
        for action in ['migrate','dev-seed']:
            subprocess.run([binary,action],cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT,check=True)
        ids=[]
        for version,(facet,label) in enumerate([('form','Tản văn'),('form','Thơ'),('theme','Ký ức'),('theme','Gia đình'),('theme','Mùa'),('theme','Đô thị')],1):
            answer=subprocess.run([binary,'tag','create',str(version),facet,label],cwd=root,env=env,text=True,capture_output=True,check=True)
            log.write(answer.stdout+answer.stderr)
            ids.append(next(line.split('\t')[1] for line in answer.stdout.splitlines() if line.startswith('tag\t')))
        for args in [['tag','alias','7',ids[2],'Hồi ức'],['tag','describe','8',ids[2],'Những điều đã qua còn ở lại trong người viết.']]:
            subprocess.run([binary,*args],cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT,check=True)
    with (out/'server.log').open('w') as log:
        server=subprocess.Popen([binary,'serve'],cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
        record['server_pid']=server.pid
        try:
            for _ in range(100):
                if server.poll() is not None:raise RuntimeError('server exited before readiness')
                try:
                    with urllib.request.urlopen('http://127.0.0.1:52567/ban-viet',timeout=2) as response:
                        if response.status==200:break
                except Exception:time.sleep(.1)
            else:raise RuntimeError('server readiness timed out')
            with (out/'browser.log').open('w') as browserlog:
                child=subprocess.Popen(command,cwd=root,env=env,stdout=browserlog,stderr=subprocess.STDOUT)
                record['browser_pid']=child.pid
                (out/'process.json').write_text(json.dumps(record,indent=2)+'\n')
                record['exit_code']=child.wait()
        finally:
            server.terminate()
            try:server.wait(timeout=15)
            except subprocess.TimeoutExpired:server.kill();server.wait()
            record['server_exit_code']=server.returncode
except Exception as error:
    record['error']=str(error);record['exit_code']=1
record['finished_utc']=datetime.datetime.now(datetime.UTC).isoformat()
(out/'result.json').write_text(json.dumps(record,indent=2)+'\n')

sys.exit(record.get('exit_code',1))

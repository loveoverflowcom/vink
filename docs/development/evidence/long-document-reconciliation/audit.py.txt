def read(path):

def sha(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT).decode().strip()


def distribution(values):
    ordered = sorted(values)
    return {
        "p50": statistics.median(ordered),
        "p95_nearest_rank": ordered[math.ceil(len(ordered) * 0.95) - 1],
    }

def rust_function(data, name):
    marker = b"pub fn " + name.encode() + b"("
    if marker not in data:
        marker = b"fn " + name.encode() + b"("
    start = data.index(marker)
    brace = data.index(b"{", start)
    depth = 0
    for index in range(brace, len(data)):
        if data[index] == 123:
            depth += 1
        elif data[index] == 125:
            depth -= 1
        if depth == 0:
            return data[start:index + 1]
    raise AssertionError("Unclosed function " + name)

        current = (ROOT / path).read_bytes()
        fingerprints[name].append({"path": path, "historical_ref": ref, "historical_sha256": sha(historical),

            fingerprints[name][-1]["only_delta"] = "Import of identical shared document_corpus; measurement helpers unchanged"
        elif path == "services/vink-app/src/domain/revision.rs":
            # #279 extracts hex formatting and adds v2. Check the v1 byte feed
            # and length prefix exactly; the strengthened runtime corpus test
            # compares all 21 current hashes/byte dimensions with #249 raw data.
            published = subprocess.check_output(["git", "show", "92b2eb33dc4ad55bfb1bed15cfb7f614e08df56c:" + path], cwd=ROOT)
            assert current == published
            before, after = rust_function(historical, "of"), rust_function(current, "of")
            assert before[:before.index(b"let digest = hasher.finalize();")] == after[:after.index(b"Self(hex_digest(hasher.finalize()))")]
            assert rust_function(historical, "push_str") == rust_function(current, "push_str")
            assert rust_function(current, "hex_digest") == b'''fn hex_digest(digest: impl AsRef<[u8]>) -> String {
    let bytes = digest.as_ref();
    let mut hex = String::with_capacity(bytes.len() * 2);
    for byte in bytes {
        hex.push_str(&format!("{byte:02x}"));
    }
    hex
}'''
            fingerprints[name][-1]["only_delta"] = "#279 adds v2; v1 byte feed and length prefix unchanged, shared lower-case hex encoder; runtime corpus oracle required"
            fingerprints[name][-1]["v1_encoding_prefix_sha256"] = sha(after[:after.index(b"Self(hex_digest(hasher.finalize()))")])
        elif path == "services/vink-app/tests/document_budget.rs":
